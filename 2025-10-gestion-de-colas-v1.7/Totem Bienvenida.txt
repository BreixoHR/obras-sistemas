        <script>
            (function(){
                var seconds = 5;
                var el = document.getElementById('timerMsg');
                function tick(){
                    if(!el) return;
                    el.textContent = seconds === 1
                        ? ('{!JSENCODE(i18n["ok_returning_in"])} 1 {!JSENCODE(i18n["second"])}...')
                        : ('{!JSENCODE(i18n["ok_returning_in"])} ' + seconds + ' {!JSENCODE(i18n["seconds"])}...');
                    seconds--;
                    if(seconds < 0){
                        // Volver a la portada de selección de idioma
                        window.location.href = window.location.pathname;
                    } else {
                        setTimeout(tick, 1000);
                    }
                }
                tick();
            })();
        </script>
