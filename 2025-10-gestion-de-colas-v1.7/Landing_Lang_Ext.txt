    private Gestion_Colas_Controller ctl;

    // ===== Exponer a VF =====
    public String  getSelectedLang()         { return selectedLangPrivate; }

    public Boolean getShowLanding()          { return showLandingPrivate; }

    // Modal de confirmación (texto y visibilidad)
    public Boolean getShowConfirm() {
        if (showLandingPrivate == true) return false;
        if (ctl == null) return false;
        try {
            List<Booking__c> ql = ctl.getQueueList(); // usar getter público del controller
            return (ql != null && !ql.isEmpty());
        } catch (Exception e) {
            return false;
        }
    }
    public String getConfirmMessage() {
        Map<String,String> i = getI18n();
        return i.containsKey('confirm_msg') ? i.get('confirm_msg')
               : 'Se le llamará por su nombre en el televisor principal';
    }

    // URL de la imagen de fondo: ?bg=https://...  ó Static Resource relativo /resource/app_bg
    public String getBgImageUrl() {
        String p = ApexPages.currentPage().getParameters().get('bg');
        if (isValidExternalUrl(p)) return p;
        // ✅ Evitar métodos retirados: usar ruta relativa del Static Resource
        return '/resource/app_bg';
    }

    // Etiqueta legible del idioma actual
    public String getLangLabel() {
        Map<String,String> labels = new Map<String,String>{
            'es' => 'Español',
            'en' => 'English',
            'fr' => 'Français',
            'de' => 'Deutsch'
        };
        String l = getLang();
        return labels.containsKey(l) ? labels.get(l) : 'Español';
    }

    // Diccionario del idioma actual para VF: {!i18n['clave']}
    public Map<String,String> getI18n() {
        String lang = getLang();
        Map<String,String> mapLang = I18N.get(lang);
        if (mapLang != null) return mapLang;
        return I18N.get('es'); // fallback
    }

    // ===== Repositorio de textos por idioma =====
    private static final Map<String, Map<String, String>> I18N = buildI18N();

    // ===== Constructores =====
    public Landing_Lang_Ext(Gestion_Colas_Controller controllerRef) {
        this.ctl = controllerRef;
        init();
    }
    public Landing_Lang_Ext() { // por si acaso
        init();
    }

    // ===== Inicialización =====
    private void init() {
        String langParam = ApexPages.currentPage().getParameters().get('lang');

        // Mostrar landing si no llega lang o si llega 'landing'
        showLandingPrivate  = String.isBlank(langParam) || 'landing'.equalsIgnoreCase(langParam);

        // Idioma por defecto: el del usuario; si llega por parámetro, úsalo
        selectedLangPrivate = normalize( String.isBlank(langParam) ? UserInfo.getLanguage() : langParam );
        if (selectedLangPrivate == 'landing') selectedLangPrivate = 'es';
    }

    // Acción invocada desde VF para elegir idioma (botones en la portada)
    public void chooseLang() {
        if (selectedLangPrivate == 'landing') {
            showLandingPrivate = true;
            selectedLangPrivate = 'es';
            return;
        }
        selectedLangPrivate = normalize(selectedLangPrivate);
        showLandingPrivate  = false; // pasamos a la vista principal
    }

    // ===== Helpers =====
    private String getLang() {
        return String.isBlank(selectedLangPrivate) ? 'es' : selectedLangPrivate;
    }

    private static String normalize(String l) {
        l = (l == null) ? 'es' : l.toLowerCase();
        if (l.startsWith('es')) return 'es';
        if (l.startsWith('en')) return 'en';
        if (l.startsWith('fr')) return 'fr';
        if (l.startsWith('de')) return 'de';
        return 'es';
    }

    private static Boolean isValidExternalUrl(String u) {
        if (String.isBlank(u)) return false;
        String lu = u.trim().toLowerCase();
        if (lu.startsWith('javascript:') || lu.startsWith('data:')) return false;
        return (lu.startsWith('http://') || lu.startsWith('https://'));
    }
